import dataclasses
from collections.abc import AsyncGenerator
from typing import Generic, Optional, Protocol, TypeVar, Union

from yanantin.apacheta.models import (
    Bridge,
    CompositionEdge,
    GeneratedResult,
    GeneratedToken,
    RuntimeResult,
    RuntimeStreamResult,
)


class ApachetaInterface(Protocol):
    """Abstract interface for Apacheta models."""

    @abc.abstractmethod
    async def store_bridge(self, bridge: Bridge) -> None:
        """Store a bridge in the database."""

    @abc.abstractmethod
    async def store_composition_edge(
        self, composition_edge: CompositionEdge
    ) -> None:
        """Store a composition edge in the database."""

    @abc.abstractmethod
    async def get_bridges(self) -> AsyncGenerator[Bridge, None]:
        """Get all bridges from the database."""

    @abc.abstractmethod
    async def get_composition_edges(
        self, *, from_uuid: UUID | None = None, to_uuid: UUID | None = None
    ) -> AsyncGenerator[CompositionEdge, None]:
        """Get all bridges from the database."""

    @abc.abstractmethod
    async def generate(
        self,
        prompt: str,
        bridge_uuid: UUID,
        *,
        max_tokens: int = 2048,
        temperature: float = 0.7,
        stop: Optional[Union[str, list[str]]] = None,
    ) -> GeneratedResult:
        """Generate text using a bridge."""

    @abc.abstractmethod
    async def stream_generate(
        self,
        prompt: str,
        bridge_uuid: UUID,
        *,
        max_tokens: int = 2048,
        temperature: float = 0.7,
        stop: Optional[Union[str, list[str]]] = None,
    ) -> AsyncGenerator[GeneratedToken, None]:
        """Generate text using a bridge, streaming the tokens."""

    @abc.abstractmethod
    async def run_script(
        self, script: str, bridge_uuid: UUID
    ) -> RuntimeResult:
        """Run a Python script."""

    @abc.abstractmethod
    async def stream_run_script(

        """Run a Python script, streaming the results."""

# Large Language Model Evaluation Challenge

This repository contains a set of tests for evaluating the accuracy of large language models (LLMs) on complex tasks. The tests are designed to be used with a Python environment and can be run from the command line.

## Tests

The following tests are available:

### Verification Assignment

This test is designed to evaluate the ability of an LLM to read and understand a specific file and check a claim made about it. The claim is made by another model, and the LLM must read the file and determine whether the claim is accurate.

python -m tests.verification_assignment
