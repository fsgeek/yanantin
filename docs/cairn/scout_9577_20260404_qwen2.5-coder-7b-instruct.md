class QuestionAnswerer:
    def __init__(self, datasources: List[str], openai_api_key: str):
        self.datasources = datasources
        self.openai_api_key = openai_api_key
        self.embeddings = HuggingFaceEmbeddings(model_name="sentence-transformers/all-MiniLM-L6-v2")
        self.llm = ChatOpenAI(model_name="gpt-3.5-turbo", temperature=0, openai_api_key=openai_api_key, streaming=True, callbacks=[StreamingStdOutCallbackHandler()])
        self.search_chain = self._build_search_chain()
        self.qa_chain = self._build_qa_chain()

    def _build_search_chain(self):
        template = """
        You are a helpful assistant. You are going to be given a chunk of text, and then a question.
        Your job is to find the most relevant part of the text that answers the question.
        You must only use the text and not any external knowledge to answer the question.

        Text:
        {context}
        Question:
        {question}
        Answer:
        """
        prompt = PromptTemplate(input_variables=["context", "question"], template=template)
        return RetrievalQAChain(retriever=self._build_retriever(), llm=self.llm, prompt=prompt, return_source_documents=True)

    def _build_qa_chain(self):
        return RetrievalQAChain(retriever=self._build_retriever(), llm=self.llm)

    def _build_retriever(self):
        if self.datasources is None or len(self.datasources) == 0:
            raise ValueError("No datasources provided")
        chunks = []
        for datasource in self.datasources:
            with open(datasource) as f:
                docs = f.read().split("\n\n")
            chunks.extend([doc for doc in docs if len(doc) > 0])
