self-similar pattern.

This command takes the size and depth of the triangle as inputs and prints
the triangle to the console.`,
	Run: func(cmd *cobra.Command, args []string) {
		size, _ := cmd.Flags().GetInt("size")
		depth, _ := cmd.Flags().GetInt("depth")
		char, _ := cmd.Flags().GetString("char")
		lines := Generate(size, depth, char)

		for _, line := range lines {

		}
	},

// Generate generates a Sierpinski triangle of the given size and depth
func Generate(size int, depth int, char string) []string {
	if len(char) == 0 {
		char = "*"
	}
	if size <= 0 {
		log.Fatal("Size must be greater than 0")
	}
	if depth < 0 {
		log.Fatal("Depth must be greater than or equal to 0")
	}

	lines := make([]string, size)
	for i := 0; i < size; i++ {
		lines[i] = strings.Repeat(" ", size-i-1) + strings.Repeat(char, i+1)
	}

	if depth == 0 {
		return lines
	}

	subLines := Generate(size/2, depth-1, char)
	for i := 0; i < size; i++ {
		lines[i] += strings.Repeat(" ", size/2) + subLines[i]
	}
	return lines
}

func getDependencyRepoPath(dependency string) (string, error) {
	cmd := exec.Command("go", "list", "-m", "-f", "{{.Dir}}", dependency)
	output, err := cmd.CombinedOutput()
	if err != nil {
		return "", fmt.Errorf("failed to get go.mod path for dependency %s: %s", dependency, err)
	}
	return strings.TrimSpace(string(output)), nil
}

func listFilesInDir(dirPath string) ([]string, error) {
	var filePaths []string

	err := filepath.Walk(dirPath, func(path string, info os.FileInfo, err error) error {
		if err != nil {
			return err
		}
		if !info.IsDir() && info.Mode().IsRegular() {
			filePaths = append(filePaths, path)
		}
		return nil
	})

	if err != nil {
		return nil, err
	}

	return filePaths, nil
}

func hasGoModFile(directory string) bool {
	_, err := os.Stat(filepath.Join(directory, "go.mod"))
	return err == nil
}

func getSubdir() (string, error) {
	cwd, err := os.Getwd()
	if err != nil {
		return "", err
	}

	var subdirs []string

	err = filepath.Walk(cwd, func(path string, info os.FileInfo, err error) error {
		if err != nil {
			return err
		}
		if info.IsDir() && hasGoModFile(path) {
			subdirs = append(subdirs, path)
		}
		return nil
	})

	if err != nil {
		return "", err
	}

	if len(subdirs) == 0 {
		return "", nil
	}

	return subdirs[0], nil
}
