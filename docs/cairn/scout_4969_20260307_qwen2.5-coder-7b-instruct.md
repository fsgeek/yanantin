        class Tensor:
            """
            A tensor is a data structure that captures a structured observation.

            Tensors are used to capture the output of a model's observation
            of a segment of codebase. The tensor structure consists of several
            parts, each of which captures a different aspect of the observation.

            The tensor structure is designed to be flexible and extensible, so
            that new parts can be added in the future without breaking existing
            code. The tensor structure is also designed to be human-readable,
            so that it can be easily inspected and understood by humans.
            """

            def __init__(
                self,
                preamble: str = "",
                strands: str = "",
                declared_losses: str = "",
                open_questions: str = "",
                closing: str = "",
            ):
                self.preamble = preamble
                self.strands = strands
                self.declared_losses = declared_losses
                self.open_questions = open_questions
                self.closing = closing

            def __repr__(self) -> str:
                return (
                    f"{self.__class__.__name__}("
                    f"preamble={repr(self.preamble)}, "
                    f"strands={repr(self.strands)}, "
                    f"declared_losses={repr(self.declared_losses)}, "
                    f"open_questions={repr(self.open_questions)}, "
                    f"closing={repr(self.closing)}"
                    f")"
                )

            def __eq__(self, o: object) -> bool:
                if not isinstance(o, Tensor):
                    return False
                return (
                    self.preamble == o.preamble and
                    self.strands == o.strands and
                    self.declared_losses == o.declared_losses and
                    self.open_questions == o.open_questions and
                    self.closing == o.closing
                )
