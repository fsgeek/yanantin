def build_file_tree(root: Path, max_depth: int = 4) -> str:
    """Build a text representation of the project file tree."""
    lines = []

    def _walk(path: Path, prefix: str, depth: int) -> None:
        if depth > max_depth:
            return
        entries = sorted(path.iterdir(), key=lambda p: (not p.is_dir(), p.name))
        # Filter noise
        skip = {
            "__pycache__", ".git", ".venv", ".uv-cache", ".serena", "node_modules",
            ".gitignore",
        }
        entries = [e for e in entries if e.name not in skip]

        for i, entry in enumerate(entries):
            is_last = i == len(entries) - 1
            connector = "--- " if is_last else "|-- "
            lines.append(f"{prefix}{connector}{entry.name}")
            if entry.is_dir():
                extension = "    " if is_last else "|   "
                _walk(entry, prefix + extension, depth + 1)

    lines.append(root.name + "/")
    _walk(root, "", 0)
    return "\n".join(lines)


def select_files_for_scout(
    root: Path, max_files: int = 8, max_lines_per_file: int = 150, coverage_map: dict | None = None,
    activity_map: dict | None = None, vantage: Path | None = None) -> list[tuple[Path, str]]:
    """Select a sample of files for the scout."""
    import random
    from typing import Dict

    def get_mod_time(path: Path) -> datetime | None:
        try:
            return path.stat().st_mtime
        except OSError:
            return None

    def get_coverage(path: Path, coverage_map: Dict[str, datetime]) -> float:
        return coverage_map.get(str(path), datetime.now(timezone.utc))

    if coverage_map is None and activity_map is None:
        candidates = [p for p in root.rglob("*") if p.is_file()]
        return [(p, str(p).split("/")[-1]) for p in candidates[:max_files]]

    candidates = [p for p in root.rglob("*") if p.is_file()]
    if not candidates:
        return []

    if coverage_map is not None:
        weights = []
        for path in candidates:
            weight = (1.0 - get_coverage(path, coverage_map) / 365.0)
            weights.append(weight)
        selected = random.choices(candidates, weights=weights, k=min(max_files, len(candidates)))
    elif activity_map is not None:
        weights = []
        for path in candidates:
            mod_time = get_mod_time(path)
            if mod_time is None:
                weights.append(0.0)
            else:
                time_diff = datetime.now(timezone.utc) - mod_time
                days = time_diff.days
                weight = max(0.0, 1.0 - days / 30.0)
                weights.append(weight)
        selected = random.choices(candidates, weights=weights, k=min(max_files, len(candidates)))
    else:
        selected = random.sample(candidates, min(max_files, len(candidates)))

    return [(p, str(p).split("/")[-1]) for p in selected]
