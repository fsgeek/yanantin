class ProofOfWork(BaseClient):
    def __init__(self, llm: BaseLLM, max_tokens=100):
        super().__init__(llm)
        self.max_tokens = max_tokens
        self.logger = logging.getLogger(__name__)

    def proof_of_work(self, question: str, proof: str, claim: Optional[str] = None) -> Union[str, None]:
        if not claim:
            claim = self.get_claim(question, proof)
        if not question:
            question = self.get_question(proof, claim)
        if not proof:
            proof = self.get_proof(question, claim)
        return proof

    def get_claim(self, question: str, proof: str) -> Union[str, None]:
        prompt = PROOF_INIT.format(proof=proof, conclusion=question)
        response = self.llm.generate(prompt, max_tokens=self.max_tokens)
        return response

    def get_question(self, proof: str, claim: str) -> Union[str, None]:
        prompt = PROOF_EVIDENCE.format(proof=proof, question=claim)
        response = self.llm.generate(prompt, max_tokens=self.max_tokens)
        return response

    def get_proof(self, question: str, claim: str) -> Union[str, None]:
        prompt = PROOF_MOTIVATION.format(question=question, conclusion=claim)
        response = self.llm.generate(prompt, max_tokens=self.max_tokens)
        return response
