def _compare(
    claims: dict[str, int | str], report: CodebaseReport
) -> list[str]:
    """Compare blueprint claims against audit reality."""
    issues: list[str] = []

    if "test_total" in claims:
        claimed = claims["test_total"]
        actual = report.test_summary.total
        if claimed != actual:
            issues.append(
