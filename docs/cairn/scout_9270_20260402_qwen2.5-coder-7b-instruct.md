        """Obfuscate a collection name."""
        raise NotImplementedError

        """Obfuscate document keys and values."""
        raise NotImplementedError

    def deobfuscate_document(self, document: dict[str, Any]) -> dict[str, Any]:
        """Deobfuscate document keys and values."""
        raise NotImplementedError
