import os
import sys
import json
import argparse
from collections import defaultdict

import requests
import tiktoken
import yaml
import yamlloader

from yanantin.utils import load_config, mk_test_tensor


def load_tensor(root: str) -> dict:
    """Load a tensor from the filesystem or ArangoDB.
    """
    try:
        with open(os.path.join(root, "tensor.json"), "r") as f:
            tensor = json.load(f)
    except FileNotFoundError:
        print(f"Tensor file not found at {os.path.join(root, 'tensor.json')}", file=sys.stderr)
        tensor = {}

    return tensor


def load_yaml(root: str) -> dict:
    """Load a YAML file from the filesystem.
    """
    yaml_path = os.path.join(root, "prose.yml")
    try:
        with open(yaml_path, 'r') as f:
            yaml_data = yaml.load(f, Loader=yamlloader.ordereddict.CLoader)
    except FileNotFoundError:
        print(f"YAML file not found at {yaml_path}", file=sys.stderr)
        yaml_data = {}

    return yaml_data


def load_tensor_and_yaml(root: str) -> dict:
    """Load a tensor and YAML file from the filesystem.
    """
    tensor = load_tensor(root)
    yaml_data = load_yaml(root)

    return tensor, yaml_data


def translate_text(text: str, config: dict) -> str:
    """Use Google Translate API to translate text to English.
    """
    url = "https://translation.googleapis.com/language/translate/v2"
    api_key = config["google_translate"]["api_key"]
    params = {
        "q": text,
        "source": "zh",
        "target": "en",
        "key": api_key
    }

    response = requests.get(url, params=params)
    if response.status_code == 200:
        data = response.json()
        translated_text = data["data"]["translations"][0]["translatedText"]
        return translated_text
    else:
        print(f"Translation failed with status code {response.status_code}")
        return text


def get_tensor_contents(tensor_root: str, config: dict) -> dict:
    """Get the contents of a tensor and its YAML file.
    """
    tensor, yaml_data = load_tensor_and_yaml(tensor_root)
    translated_tensor = {
        "author": translate_text(tensor["author"], config),
        "context": translate_text(tensor["context"], config),
        "predecessors_in_scope": tensor.get("predecessors_in_scope", []),
        "prose": translate_text(tensor["prose"], config)
    }

    return translated_tensor, yaml_data


def get_tensor_tree(root: str, config: dict) -> dict:
    """Get the tensor tree, including tensor contents and YAML data.

    Args:
        root (str): The root directory of the tensor tree.

    Returns:
        dict: A dictionary representing the tensor tree.
    """
    tensor_tree = {}
    for dirpath, dirnames, filenames in os.walk(root):
        for filename in filenames:
            if filename == "tensor.json":
                tensor_root = os.path.join(dirpath, os.path.pardir)
                translated_tensor, yaml_data = get_tensor_contents(tensor_root, config)
                tensor_tree[tensor_root] = {
                    "tensor": translated_tensor,
                    "yaml": yaml_data
                }

    return tensor_tree


def main(args):
    """Translate tensor contents for all tensors under the specified root directory.

    tensor_tree = get_tensor_tree(args.root, config)

if __name__ == "__main__":
