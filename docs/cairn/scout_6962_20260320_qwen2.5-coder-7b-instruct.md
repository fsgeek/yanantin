class OpenAIError(Exception):
    def __init__(self, message: str, error: openai_error.OpenAIError):
        self.message = message
        self.error = error
        super().__init__(message)


class OpenAI:
    def __init__(self, model: str = "gpt-3.5-turbo", temperature: float = 0.0, max_tokens: int = 1000):
        self.model = model
        self.temperature = temperature
        self.max_tokens = max_tokens

    def __call__(self, messages: list[Dict[str, str]], verbose: bool = False) -> ChatCompletion:
        try:
            response = openai.ChatCompletion.create(
                model=self.model,
                messages=messages,
                temperature=self.temperature,
                max_tokens=self.max_tokens,
            )
            if verbose:
                log.debug(f"OpenAI response: {response}")
            return response
        except openai_error.OpenAIError as e:
            raise OpenAIError("OpenAI error", e)

    @staticmethod
    def prompt(prompt: str, messages: list[Dict[str, str]], verbose: bool = False) -> str:
        messages.append({"role": "user", "content": prompt})
        response = OpenAI().messages(messages, verbose)
        response = OpenAI._extract_response(response)
        return response
