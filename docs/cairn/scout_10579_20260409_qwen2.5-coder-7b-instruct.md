```
"""Red-bar test: Composition associative invariant.

If a tensor has two parents, and both parents have the same parent, then those
parents are the same tensor. (This is an associative property of composition.)
"""

from yanantin.apacheta.backends.memory import InMemoryBackend
from yanantin.apacheta.models import (
    CompositionEdge,
    TensorRecord,
)
from yanantin.composition import (
    compose_tensors,
    infer_parents,
)
from uuid import uuid4


def test_associative_composition():
    """Composing 2 parents, then composing those parents with a third, is the same
    as composing the 3 parents directly.
    """
    backend = InMemoryBackend()
    tensor_a = TensorRecord(id=uuid4(), provenance=None)
    tensor_b = TensorRecord(id=uuid4(), provenance=None)
    tensor_c = TensorRecord(id=uuid4(), provenance=None)
    backend.store_tensor(tensor_a)
    backend.store_tensor(tensor_b)
    backend.store_tensor(tensor_c)

    # tensor_a and tensor_b have the same parent
    parent_tensor = TensorRecord(id=uuid4(), provenance=None)
    backend.store_tensor(parent_tensor)
    edge_a = CompositionEdge(
        from_tensor=tensor_a.id,
        to_tensor=parent_tensor.id,
        relation_type="COMPOSES_WITH",
    )
    backend.store_composition_edge(edge_a)
    edge_b = CompositionEdge(
        from_tensor=tensor_b.id,
        to_tensor=parent_tensor.id,
        relation_type="COMPOSES_WITH",
    )
    backend.store_composition_edge(edge_b)

    # tensor_a, tensor_b, and tensor_c have the same parent
    grandparent_tensor = TensorRecord(id=uuid4(), provenance=None)
    backend.store_tensor(grandparent_tensor)
    edge_c = CompositionEdge(
        from_tensor=tensor_c.id,
        to_tensor=grandparent_tensor.id,
        relation_type="COMPOSES_WITH",
    )
    backend.store_composition_edge(edge_c)

    # Compose tensor_a and tensor_b, then compose that result with tensor_c
    # (this should be the same as composing tensor_a, tensor_b, and tensor_c
    # directly)
    a_b = compose_tensors([tensor_a, tensor_b], backend=backend)
    result = compose_tensors([a_b, tensor_c], backend=backend)
    expected = compose_tensors([tensor_a, tensor_b, tensor_c], backend=backend)
    assert result == expected

    # Verify that the interfaces to the tensors are correct
    assert infer_parents(result)[0] == grandparent_tensor.id
    assert infer_parents(result)[1] == parent_tensor.id

```

    def get_tensor(self, tensor_id: str) -> TensorRecord:
        """Get a tensor from the backend."""
        try:
            return self.tensors[tensor_id]
        except KeyError:
            raise KeyError(f"Tensor {tensor_id} not found in memory backend")

    def store_tensor(self, tensor: TensorRecord) -> None:
        """Store a tensor in the backend."""
        if tensor.id in self.tensors:
            raise ValueError(f"Tensor {tensor.id} already exists in memory backend")
        self.tensors[tensor.id] = tensor

    def store_composition_edge(self, edge: CompositionEdge) -> None:
        """Store a composition edge in the backend."""
        if not edge.from_tensor or not edge.to_tensor:
            raise ValueError("Cannot store edge with empty to or from tensor")
        if edge.from_tensor not in self.tensors:
            raise ValueError(f"From tensor {edge.from_tensor} does not exist")
        if edge.to_tensor not in self.tensors:
            raise ValueError(f"To tensor {edge.to_tensor} does not exist")
        if not edge.relation_type:
            raise ValueError("Cannot store edge with empty relation type")
        self.edges.setdefault((edge.from_tensor, edge.to_tensor), []).append(
            (edge.from_tensor, edge.to_tensor, edge.relation_type)
        )

        self,
