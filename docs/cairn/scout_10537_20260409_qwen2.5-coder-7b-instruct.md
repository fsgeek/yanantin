def parse_changelog(content: str) -> List[str]:
    """
    Parse the changelog from the content and return the list of changelog entries.
    Each changelog entry is a list of lines.
    """
    changelog_pattern = re.compile(r"## \d+\.\d+\.\d+\n([\s\S]*?)(?=\n## \d+\.\d+\.\d+|\n\d+\.\d+\.\d+\n)")
    changelog_entries = changelog_pattern.findall(content)
    return [entry.split("\n") for entry in changelog_entries]


def filter_changelog_entries(entries: List[List[str]]) -> List[List[str]]:
    """
    Remove entries that don't contain "Removal" and "Addition".
    """
    return [
        entry
        for entry in entries
        if any(("Removal" in line or "Addition" in line) for line in entry)
    ]
