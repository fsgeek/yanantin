def run_collector_cli(*args: str) -> subprocess.CompletedProcess[str]:
    """Invoke ``python -m yanantin.collector`` with the given args."""
    cmd = [sys.executable, "-m", "yanantin.collector", *args]

    pythonpath = env.get("PYTHONPATH")
    combined = [str(SRC_PATH)]

def main() -> None:
    """Collects metadata about the current environment."""
    parser = argparse.ArgumentParser(description=__doc__)
    parser.add_argument("--json", help="whether to output the result as JSON", action="store_true")
    parser.add_argument("--record", help="whether to record the result as a tensor", action="store_true")
    parser.add_argument("--version", help="show the version", action="store_true")
    args = parser.parse_args()

    if args.version:
        print(__version__)
        return

    if args.json:
        payload = collect()
        print(json.dumps(payload))
        return

    if args.record:
        result = collect()
        tensor_id = result["tensor_id"]
        print(f"Recorded as tensor {tensor_id}")
        return

    print("Yanantin Collector")
    print("Machine Configuration")
    print(collect(), end="")
    sys.exit(0)
