def test_claim_scour_number(temp_cairn):
    """Test the Lamport bakery for unique number allocation."""
    run_number, path = _claim_scour_number(temp_cairn, "test_model")
    assert path.name == "scour_0001_20230808_test_model.md"
    assert run_number == 1

    run_number, path = _claim_scour_number(temp_cairn, "test_model")
    assert path.name == "scour_0002_20230808_test_model.md"
    assert run_number == 2

    # Test with existing files
    temp_file = temp_cairn / "scour_0002_20230808_test_model.md"
    temp_file.touch()
    run_number, path = _claim_scour_number(temp_cairn, "test_model")
    assert path.name == "scour_0003_20230808_test_model.md"
    assert run_number == 3

def test_is_degenerate_repetition():
    """Test detection of degenerate repetition loops."""
    assert not _is_degenerate_repetition("abc" * 100)
    assert not _is_degenerate_repetition("abcab" * 100)
    assert _is_degenerate_repetition("abc" * 1000)
    assert _is_degenerate_repetition("abcab" * 1000)
    assert not _is_degenerate_repetition("abcab" * 100 + "def" * 100)
    assert _is_degenerate_repetition("abc" * 100 + "def" * 100, threshold=2)
